1//! # Bin Counts Module
2//!
3//! This module computes a shared genomic bin grid across one or more BAM files and
4//! counts filtered reads (or fragments) per bin per sample. It supports:
5//! *   A single shared bin grid across samples, required for between-sample comparison.
6//! *   Parallel per-chunk counting, reusing the same binning machinery as `coverage_analysis`.
7//! *   Conversion of the resulting count matrix to a Polars `DataFrame` for QC / M-A plots.
8
9use std::path::PathBuf;
10
11use crate::bam_utils::{BamStats, Iv, bin_intervals, get_bam_header, is_scaffold, progress_bar};

23/// One genomic bin in the count matrix.
24///
25/// Coordinates are 1-based (noodles/SAM convention), matching `BamStats`.
26#[derive(Debug, Clone)]
27pub struct BinRegion {
28    /// Chromosome / reference sequence name.
29    pub chrom: String,
30    /// 1-based inclusive start position.
31    pub start: u64,
32    /// 1-based inclusive end position.
33    pub end: u64,
34}
35
36/// A sample × bin read-count matrix, produced by [`count_bins`].
37pub struct BinCounts {
38    /// Sample names, in column order matching `counts`.
39    pub sample_names: Vec<String>,
40    /// The bin size used to build the grid.
41    pub bin_size: u64,
42    /// The bin grid, in row order matching `counts`.
43    pub bins: Vec<BinRegion>,
44    /// Read counts, shape `(n_bins, n_samples)`.
45    pub counts: Array2<u64>,
46    /// Post-filter library size (total retained reads/fragments) per sample.
47    pub library_sizes: Vec<u64>,
48}
49
50impl BinCounts {
51    /// Convert the bin count matrix into a Polars `DataFrame` for QC (e.g. M-A plots).
52    ///
53    /// # Returns
54    ///
55    /// A `DataFrame` with columns `chrom`, `start`, `end`, and one `f64` column per sample,
56    /// named after the corresponding entry in `sample_names`.
57    pub fn to_polars(&self) -> Result<DataFrame> {
58        let n_bins = self.bins.len();
59        let mut chroms = Vec::with_capacity(n_bins);
60        let mut starts = Vec::with_capacity(n_bins);
61        let mut ends = Vec::with_capacity(n_bins);
62        for bin in &self.bins {
63            chroms.push(bin.chrom.clone());
64            starts.push(bin.start);
65            ends.push(bin.end);
66        }
67
68        let mut columns = vec![
69            Column::new("chrom".into(), chroms),
70            Column::new("start".into(), starts),
71            Column::new("end".into(), ends),
72        ];
73
74        for (s, name) in self.sample_names.iter().enumerate() {
75            let col: Vec<f64> = self.counts.column(s).iter().map(|&c| c as f64).collect();
76            columns.push(Column::new(name.as_str().into(), col));
77        }
78
79        Ok(DataFrame::new(columns)?)
80    }
81}

83/// Bin-count reads (or fragments) across one or more BAM files onto a shared genomic bin grid.
84///
85/// All input BAM files must share an identical chromosome name → length mapping; this is
86/// required so that a single bin grid can be used to compare samples bin-for-bin.
87///
88/// # Arguments
89///
90/// * `bam_paths` - Input BAM files, one per sample.
91/// * `sample_names` - Sample names, same length and order as `bam_paths`.
92/// * `bin_size` - Genomic bin width in base pairs.
93/// * `filters` - Per-sample read filter, same length and order as `bam_paths`.
94/// * `use_fragment` - Count fragments (read pairs) instead of individual read alignments.
95/// * `ignore_scaffold_chromosomes` - Skip scaffold / unplaced chromosomes when building the grid.
96///
97/// # Returns
98///
99/// A [`BinCounts`] with the count matrix and per-sample post-filter library sizes.
100pub fn count_bins(
101    bam_paths: &[PathBuf],
102    sample_names: &[String],
103    bin_size: u64,
104    filters: &[BamReadFilter],
105    use_fragment: bool,
106    ignore_scaffold_chromosomes: bool,
107) -> Result<BinCounts> {
108    anyhow::ensure!(
109        !bam_paths.is_empty(),
110        "At least one BAM file is required for bin counting"
111    );
112    anyhow::ensure!(
113        bam_paths.len() == sample_names.len(),
114        "Number of BAM paths ({}) does not match number of sample names ({})",
115        bam_paths.len(),
116        sample_names.len()
117    );
118    anyhow::ensure!(
119        bam_paths.len() == filters.len(),
120        "Number of BAM paths ({}) does not match number of filters ({})",
121        bam_paths.len(),
122        filters.len()
123    );
124
125    // Build per-sample BamStats and validate a shared chromosome grid.
126    let bam_stats: Vec<BamStats> = bam_paths
127        .iter()
128        .map(|p| {
129            BamStats::new(p.clone())
130                .with_context(|| format!("Failed to read BAM stats for {}", p.display()))
131        })
132        .collect::<Result<Vec<_>>>()?;
133
134    let reference_chromsizes = bam_stats[0].chromsizes_ref_name()?;
135    for (stats, path) in bam_stats.iter().zip(bam_paths.iter()).skip(1) {
136        let chromsizes = stats.chromsizes_ref_name()?;
137        if chromsizes != reference_chromsizes {
138            bail!(
139                "Chromosome name/length mismatch between BAM files: {} does not match {}. \
140                 bam-normalize requires all input BAM files to share an identical set of \
141                 reference sequences so that a common bin grid can be used.",
142                path.display(),
143                bam_paths[0].display()
144            );
145        }
146    }
147
148    // Build the global bin grid: one contiguous block of bins per chromosome, in
149    // deterministic (sorted) chromosome name order.
150    let mut chrom_names: Vec<String> = reference_chromsizes.keys().cloned().collect();
151    if ignore_scaffold_chromosomes {
152        chrom_names.retain(|name| !is_scaffold(name));
153    }
154    chrom_names.sort();
155
156    let mut bins = Vec::new();
157    let mut chrom_offsets: HashMap<String, usize> = HashMap::default();
158    for chrom in &chrom_names {
159        chrom_offsets.insert(chrom.clone(), bins.len());
160        let length = reference_chromsizes[chrom];
161        let mut start = 1u64;
162        while start <= length {
163            let end = (start + bin_size - 1).min(length);
164            bins.push(BinRegion {
165                chrom: chrom.clone(),
166                start,
167                end,
168            });
169            start = end + 1;
170        }
171    }
172    let n_bins = bins.len();
173    anyhow::ensure!(
174        n_bins > 0,
175        "No bins produced; check --bin-size and --ignore-scaffolds"
176    );
177
178    let mut counts = Array2::<u64>::zeros((n_bins, bam_paths.len()));
179    let mut library_sizes = Vec::with_capacity(bam_paths.len());
180
181    for (s, ((bam_path, stats), filter)) in bam_paths
182        .iter()
183        .zip(bam_stats.iter())
184        .zip(filters.iter())
185        .enumerate()
186    {
187        let header = get_bam_header(bam_path.clone())?;
188        let chromsizes_refid = stats.chromsizes_ref_id()?;
189        let genomic_chunks: Vec<_> = stats
190            .genome_chunks(bin_size)?
191            .into_iter()
192            .filter(|region| {
193                if !ignore_scaffold_chromosomes {
194                    return true;
195                }
196                let name = region.name().to_owned().to_string();
197                !is_scaffold(&name)
198            })
199            .collect();
200        let n_total_chunks = genomic_chunks.len();
201
202        let chunk_results: Vec<(String, usize, u32)> = genomic_chunks
203            .into_par_iter()
204            .progress_with(progress_bar(
205                n_total_chunks as u64,
206                format!("Counting bins for {}", sample_names[s]),
207            ))
208            .map(|region| -> Result<Vec<(String, usize, u32)>> {
209                let mut reader = bam::io::indexed_reader::Builder::default()
210                    .build_from_path(bam_path)
211                    .context("Failed to open BAM file")?;
212                let records = reader.query(&header, &region)?;
213                let intervals: Vec<Iv> = records
214                    .records()
215                    .filter_map(|record| record.ok())
216                    .filter_map(|record| {
217                        IntervalMaker::new(
218                            record,
219                            &header,
220                            &chromsizes_refid,
221                            filter,
222                            use_fragment,
223                            None,
224                            None,
225                        )
226                        .coords()
227                        .map(|(start_pos, end_pos, _dtlen)| Iv {
228                            start: start_pos,
229                            stop: end_pos,
230                            val: 1,
231                        })
232                    })
233                    .collect();
234
235                let lapper = Lapper::new(intervals);
236                let region_interval = region.interval();
237                let region_start = region_interval
238                    .start()
239                    .context("Failed to get region start")?
240                    .get();
241                let region_end = region_interval
242                    .end()
243                    .context("Failed to get region end")?
244                    .get();
245
246                let chrom_name = region.name().to_owned().to_string();
247                let bin_counts = bin_intervals(&lapper, region_start, region_end, bin_size, false);
248                Ok(bin_counts
249                    .into_iter()
250                    .map(|iv| (chrom_name.clone(), iv.start, iv.val))
251                    .collect())
252            })
253            .fold(
254                || Ok(Vec::new()),
255                |acc: Result<Vec<(String, usize, u32)>>,
256                 result: Result<Vec<(String, usize, u32)>>| {
257                    let mut acc = acc?;
258                    acc.extend(result?);
259                    Ok(acc)
260                },
261            )
262            .reduce(
263                || Ok(Vec::new()),
264                |acc: Result<Vec<(String, usize, u32)>>, b: Result<Vec<(String, usize, u32)>>| {
265                    let mut acc = acc?;
266                    acc.extend(b?);
267                    Ok(acc)
268                },
269            )?;
270
271        for (chrom, start, count) in chunk_results {
272            let offset = *chrom_offsets.get(&chrom).ok_or_else(|| {
273                anyhow::anyhow!("Chromosome {chrom} not part of the shared bin grid")
274            })?;
275            let bin_idx_in_chrom = ((start as u64 - 1) / bin_size) as usize;
276            counts[[offset + bin_idx_in_chrom, s]] = count as u64;
277        }
278
279        library_sizes.push(filter.stats().n_reads_after_filtering());
280    }
281
282    Ok(BinCounts {
283        sample_names: sample_names.to_vec(),
284        bin_size,
285        bins,
286        counts,
287        library_sizes,
288    })
289}

291#[cfg(test)]
292mod tests {
293    use super::*;
294    use crate::read_filter::BamReadFilter;

296    fn get_test_data_dir() -> PathBuf {
297        PathBuf::from(env!("CARGO_MANIFEST_DIR"))
298            .parent()
299            .expect("Failed to get workspace root")
300            .join("test/data")
301    }
302
303    #[test]
304    fn test_count_bins_same_bam_twice_gives_identical_columns() {
305        let bam_file = get_test_data_dir().join("test.bam");
306        let bam_paths = vec![bam_file.clone(), bam_file];
307        let sample_names = vec!["a".to_string(), "b".to_string()];
308        let filters = vec![BamReadFilter::default(), BamReadFilter::default()];
309
310        let counts = count_bins(&bam_paths, &sample_names, 10_000, &filters, false, true)
311            .expect("count_bins failed");
312
313        assert!(!counts.bins.is_empty(), "Expected at least one bin");
314        assert_eq!(counts.library_sizes[0], counts.library_sizes[1]);
315        assert!(counts.library_sizes[0] > 0);
316
317        let total: u64 = counts.counts.iter().sum();
318        assert!(total > 0, "Expected non-zero counts");
319
320        for row in counts.counts.rows() {
321            assert_eq!(row[0], row[1]);
322        }
323    }
324
325    #[test]
326    fn test_count_bins_mismatched_bam_paths_and_sample_names_errors() {
327        let bam_file = get_test_data_dir().join("test.bam");
328        let bam_paths = vec![bam_file];
329        let sample_names = vec!["a".to_string(), "b".to_string()];
330        let filters = vec![BamReadFilter::default()];
331
332        let result = count_bins(&bam_paths, &sample_names, 10_000, &filters, false, true);
333        assert!(result.is_err());
334    }
