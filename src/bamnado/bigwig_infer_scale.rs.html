9/// Read buffer per thread — large enough to amortise syscall overhead on whole-chromosome fetches.
10const BW_BUF_BYTES: usize = 2 << 20; // 2 MB
11
12fn open_bw(path: &Path) -> Result<bigtools::BigWigRead<BufReader<File>>> {
13    let file = File::open(path).with_context(|| format!("Cannot open {}", path.display()))?;
14    let buf = BufReader::with_capacity(BW_BUF_BYTES, file);
15    bigtools::BigWigRead::open(buf).map_err(|e| anyhow::anyhow!("BigWig open error: {e}"))
16}

18fn ser_f64_max_null<S>(v: &f64, s: S) -> Result<S::Ok, S::Error>
19where
20    S: serde::Serializer,
21{
22    if *v == f64::MAX {
23        s.serialize_none()
24    } else {
25        s.serialize_some(v)
26    }
27}

29fn ser_f64_nan_null<S>(v: &f64, s: S) -> Result<S::Ok, S::Error>
30where
31    S: serde::Serializer,
32{
33    if v.is_nan() {
34        s.serialize_none()
35    } else {
36        s.serialize_some(v)
37    }
38}

40/// Tuning knobs for `infer_scale_factor`.
41#[derive(Debug, Clone)]
42pub struct InferScaleConfig {
43    /// Stop scanning a chromosome after this many consecutive intervals with no new minimum.
44    ///
45    /// The heuristic is safe when the minimum bin value appears within the first
46    /// `min_stable_streak` intervals of a chromosome.  For real normalised BigWigs
47    /// (CPM/RPKM) single-read bins are scattered throughout the genome, so 50 000 is
48    /// conservative.  Set to `u32::MAX` to disable early exit and always do a full scan.
49    ///
50    /// **Limitation**: if ALL minimum bins are clustered at the very end of a chromosome
51    /// and the chromosome also contains two or more distinct larger values before that
52    /// point, the heuristic may report a higher minimum.  This is unlikely in practice
53    /// but can be demonstrated with synthetic data — see `test_heuristic_misses_late_minimum`.
54    pub min_stable_streak: u32,
55
56    /// Maximum number of interval start positions collected for bin-size (GCD) inference.
57    ///
58    /// The GCD of bin-aligned start positions stabilises within the first few entries.
59    /// Capping avoids O(intervals) `HashSet` growth on high-depth files where nearly
60    /// every bin is non-zero.
61    pub max_starts: usize,
62}
63
64impl Default for InferScaleConfig {
65    fn default() -> Self {
66        Self {
67            min_stable_streak: 50_000,
68            max_starts: 512,
69        }
70    }
71}

73/// Inferred normalisation method.
74#[derive(Debug, Clone, PartialEq, Serialize)]
75pub enum NormMethod {
76    Cpm,
77    Rpkm,
78    /// Both CPM and RPKM give a plausible library size — cannot distinguish.
79    Ambiguous,
80    /// Neither CPM nor RPKM gives a plausible library size (RPGC, BPM, raw, …).
81    Unknown,
82}
83
84impl std::fmt::Display for NormMethod {
85    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
86        match self {
87            NormMethod::Cpm => write!(f, "CPM"),
88            NormMethod::Rpkm => write!(f, "RPKM"),
89            NormMethod::Ambiguous => write!(f, "Ambiguous (CPM or RPKM)"),
90            NormMethod::Unknown => write!(f, "Unknown"),
91        }
92    }
93}

95#[derive(Debug)]
96pub struct Warnings {
97    /// More than 2 distinct interval widths — variable bin sizes found.
98    pub variable_bin_sizes: bool,
99    /// second_min / min is not near an integer — smoothing / kernel applied.
100    pub smoothing_detected: bool,
101    /// ratio > 2 and near integer — pseudocount shifted the signal floor.
102    pub pseudocount_detected: bool,
103    /// Minimum value came from a chromosome that is < 0.1 % of genome size.
104    pub min_from_small_chrom: bool,
105}
106
107impl Warnings {
108    /// Returns active warnings as human-readable strings (empty when no warnings).
109    pub fn messages(&self) -> Vec<&'static str> {
110        let mut v = Vec::new();
111        if self.variable_bin_sizes {
112            v.push("variable_bin_sizes: mixed bin widths detected, RPKM reversal unreliable");
113        }
114        if self.smoothing_detected {
115            v.push("smoothing_detected: ratio non-integer, recovery is approximate");
116        }
117        if self.pseudocount_detected {
118            v.push("pseudocount_detected: scale factor corrected to second_min");
119        }
120        if self.min_from_small_chrom {
121            v.push(
122                "min_from_small_chrom: minimum from small/unplaced chromosome, may be artefactual",
123            );
124        }
125        v
126    }
127}

135#[derive(Debug, Serialize)]
136pub struct InferScaleResult {
137    /// Inferred normalisation method.
138    pub norm_method: NormMethod,
139    /// Scale factor s such that raw ≈ normalised × s.
140    pub scale_factor: f64,
141    /// Implied library size (total mapped reads or read-pairs).
142    pub library_size: f64,
143    /// Canonical bin size (bp) inferred from interval widths.
144    pub bin_size: u32,
145    /// Global minimum non-zero bin value.
146    pub min_val: f64,
147    /// Second-smallest distinct non-zero bin value (null when only one distinct value seen).
148    #[serde(serialize_with = "ser_f64_max_null")]
149    pub second_min_val: f64,
150    /// second_min / min (diagnostic ratio; null when second_min unknown).
151    #[serde(serialize_with = "ser_f64_nan_null")]
152    pub ratio: f64,
153    /// Estimated pseudocount (if detected).
154    pub pseudocount: Option<f64>,
155    /// True when the confidence criterion was met before exhausting chromosomes.
156    pub confident: bool,
157    pub warnings: Warnings,
158    /// Number of chromosomes scanned.
159    pub chroms_scanned: usize,
160}
161
162impl InferScaleResult {
163    /// Convert a normalised BigWig value back to raw read count per bin.
164    ///
165    /// raw = value × scale_factor
166    ///
167    /// Derivation:
168    ///   CPM:  value = reads / (N/1e6)       →  raw = value × (N/1e6)
169    ///   RPKM: value = reads / (N×bin/1e9)   →  raw = value × (N×bin/1e9)
170    ///   Both reduce to: raw = value × scale_factor
171    ///
172    /// When a pseudocount was detected the scale_factor already uses second_min
173    /// (the smallest bin with ≥1 real read) so raw counts remain integer-valued.
174    pub fn to_raw(&self, value: f64) -> f64 {
175        value * self.scale_factor
176    }
177}

179/// Infer scale factor and normalisation method from a BigWig file.
180///
181/// Chromosomes are scanned in three priority groups (mid-sized autosomes first,
182/// large chromosomes second, small/unplaced last).  Each group is processed in
183/// parallel — every worker thread opens its own buffered reader.  Scanning stops
184/// as soon as the minimum value has been confirmed on ≥2 chromosomes and the
185/// second_min/min ratio is near an integer.
186pub fn infer_scale_factor(bw_path: &Path, config: &InferScaleConfig) -> Result<InferScaleResult> {
187    let chrom_info: Vec<bigtools::ChromInfo> = {
188        let reader = open_bw(bw_path)?;
189        reader.chroms().to_owned()
190    };
191
192    if chrom_info.is_empty() {
193        bail!("BigWig file has no chromosomes");
194    }
195
196    let genome_size: u64 = chrom_info.iter().map(|c| c.length as u64).sum();
197
198    // Partition into three priority groups — mid-sized autosomes first because
199    // they are the most likely to share the global minimum and confirm confidence
200    // quickly, while being small enough to scan fast.
201    let mut groups: [Vec<bigtools::ChromInfo>; 3] = [vec![], vec![], vec![]];
202    for c in &chrom_info {
203        groups[chrom_priority(&c.name, c.length, genome_size) as usize].push(c.clone());
204    }
205
206    let mut global_min = f64::MAX;
207    let mut global_second_min = f64::MAX;
208    let mut global_starts: HashSet<u32> = HashSet::new();
209    let mut min_confirmations: u32 = 0;
210    let mut chroms_scanned: usize = 0;
211    let mut confident = false;
212    let mut min_chrom_name = String::new();
213
214    'outer: for group in &groups {
215        if group.is_empty() || confident {
216            continue;
217        }
218
219        // All chromosomes in this priority group are scanned in parallel.
220        // Each closure opens its own BigWigRead so there is no shared I/O state.
221        type ChromScan = Result<(String, f64, f64, HashSet<u32>)>;
222        let scan_results: Vec<ChromScan> = group
223            .par_iter()
224            .map(|chrom| {
225                let mut r = open_bw(bw_path)?;
226                let (cmin, c2min, cwidths) = scan_chrom(&mut r, &chrom.name, chrom.length, config)?;
227                Ok((chrom.name.clone(), cmin, c2min, cwidths))
228            })
229            .collect();
230
231        // Merge results sequentially — order within a group doesn't matter for
232        // correctness; the priority ordering ensures the most informative group
233        // is processed first so confidence is reached with fewest total scans.
234        for res in scan_results {
235            let (chrom_name, cmin, c2min, cwidths) = res?;
236            chroms_scanned += 1;
237            global_starts.extend(cwidths);
238
239            if cmin == f64::MAX {
240                continue;
241            }
242
243            const EPS: f64 = 1e-9;
244
245            if cmin < global_min - EPS {
246                let old_min = global_min;
247                let old_second = global_second_min;
248                global_second_min = [old_min, old_second, c2min]
249                    .iter()
250                    .copied()
251                    .filter(|&v| v > cmin + EPS)
252                    .fold(f64::MAX, f64::min);
253                global_min = cmin;
254                min_chrom_name = chrom_name;
255                min_confirmations = 1;
256            } else if (cmin - global_min).abs() < EPS {
257                min_confirmations += 1;
258                if c2min < global_second_min - EPS {
259                    global_second_min = c2min;
260                }
261            } else {
262                if cmin < global_second_min - EPS {
263                    global_second_min = cmin;
264                }
265                if c2min > global_min + EPS && c2min < global_second_min - EPS {
266                    global_second_min = c2min;
267                }
268            }
269
270            if min_confirmations >= 2 && global_second_min < f64::MAX {
271                let ratio = global_second_min / global_min;
272                if (ratio - ratio.round()).abs() < 0.01 {
273                    confident = true;
274                    break 'outer;
275                }
276            }
277        }
278    }
279
280    if global_min == f64::MAX {
281        bail!("No non-zero values found in BigWig file");
282    }
283
284    // GCD of all observed widths (end-truncated bins are excluded in scan_chrom).
285    // GCD = 0 → no widths collected; GCD = 1 → no common factor → truly mixed bin sizes.
286    let canonical_bin_size = global_starts.iter().copied().fold(0u32, gcd);
287    let variable_bin_sizes = canonical_bin_size <= 1;
288
289    let ratio = if global_second_min < f64::MAX {
290        global_second_min / global_min
291    } else {
292        f64::NAN
293    };
294
295    let smoothing_detected = !ratio.is_nan() && (ratio - ratio.round()).abs() >= 0.01;
296
297    // Pseudocount: ratio > 2 and near integer.
298    // Min bin = pseudocount-only bin (0 raw reads). s_corrected = 1 / second_min.
299    let (pseudocount, scale_factor) =
300        if !ratio.is_nan() && ratio > 2.0 + 0.05 && (ratio - ratio.round()).abs() < 0.05 {
301            let p = 1.0 / (ratio - 1.0);
302            (Some(p), 1.0 / global_second_min)
303        } else {
304            (None, 1.0 / global_min)
305        };
306
307    // CPM: min_val = 1e6 / N  →  N = 1e6 / min_val
308    // RPKM: min_val = 1e9 / (N × bin_size)  →  N = 1e9 / (min_val × bin_size)
309    let n_cpm = 1e6 / global_min;
310    let n_rpkm = if canonical_bin_size > 0 {
311        1e9 / (global_min * canonical_bin_size as f64)
312    } else {
313        f64::NAN
314    };
315
316    let plausible = |n: f64| (1e6..=2e9).contains(&n);
317    let norm_method = match (plausible(n_cpm), !n_rpkm.is_nan() && plausible(n_rpkm)) {
318        (true, false) => NormMethod::Cpm,
319        (false, true) => NormMethod::Rpkm,
320        (true, true) => NormMethod::Ambiguous,
321        (false, false) => NormMethod::Unknown,
322    };
323
324    let library_size = match norm_method {
325        NormMethod::Rpkm => n_rpkm,
326        _ => n_cpm,
327    };
328
329    let min_chrom_fraction = chrom_info
330        .iter()
331        .find(|c| c.name == min_chrom_name)
332        .map(|c| c.length as f64 / genome_size as f64)
333        .unwrap_or(1.0);
334    let min_from_small_chrom = min_chrom_fraction < 0.001;
335
336    Ok(InferScaleResult {
337        norm_method,
338        scale_factor,
339        library_size,
340        bin_size: canonical_bin_size,
341        min_val: global_min,
342        second_min_val: global_second_min,
343        ratio,
344        pseudocount,
345        confident,
346        warnings: Warnings {
347            variable_bin_sizes,
348            smoothing_detected,
349            pseudocount_detected: pseudocount.is_some(),
350            min_from_small_chrom,
351        },
352        chroms_scanned,
353    })
354}

356fn gcd(a: u32, b: u32) -> u32 {
357    if b == 0 { a } else { gcd(b, a % b) }
358}

360fn scan_chrom<R: Read + Seek>(
361    reader: &mut bigtools::BigWigRead<R>,
362    chrom: &str,
363    chrom_len: u32,
364    config: &InferScaleConfig,
365) -> Result<(f64, f64, HashSet<u32>)> {
366    let it = reader.get_interval(chrom, 0, chrom_len)?;
367    let mut min1 = f64::MAX;
368    let mut min2 = f64::MAX;
369    let mut starts: HashSet<u32> = HashSet::new();
370    // Counts intervals since the last time min1 changed.  Once this exceeds
371    // min_stable_streak we have seen enough of the distribution to trust min1/min2.
372    //
373    // IMPORTANT: the break only fires when min2 < f64::MAX (i.e. we have already
374    // seen two distinct values), which prevents exiting on a single-value plateau.
375    // However, if a lower minimum exists beyond the streak window AND the chromosome
376    // already has two distinct values, it will be missed.  Use min_stable_streak =
377    // u32::MAX to disable this optimisation and always do a full scan.
378    let mut stable_streak: u32 = 0;
379
380    for r in it {
381        let iv = r?;
382        if iv.value <= 0.0 || iv.end <= iv.start {
383            continue;
384        }
385        let v = iv.value as f64;
386
387        // Collect start positions for bin_size inference via GCD.
388        // GCD stabilises within the first few multiples-of-bin_size entries,
389        // so stop inserting once we have enough — avoids O(intervals) HashSet growth
390        // on high-depth files where most bins are non-zero.
391        if iv.start > 0 && starts.len() < config.max_starts {
392            starts.insert(iv.start);
393        }
394
395        if v < min1 - 1e-9 {
396            if min1 < min2 {
397                min2 = min1;
398            }
399            min1 = v;
400            stable_streak = 0;
401        } else {
402            if v > min1 + 1e-9 && v < min2 {
403                min2 = v;
404            }
405            stable_streak += 1;
406            if stable_streak >= config.min_stable_streak && min2 < f64::MAX {
407                break;
408            }
409        }
410    }
411
412    Ok((min1, min2, starts))
413}

415/// Returns sort key: 0 = mid-sized autosomes, 1 = large, 2 = small/unplaced/mito.
416pub(crate) fn chrom_priority(name: &str, len: u32, genome_size: u64) -> u8 {
417    if genome_size == 0 {
418        return 1;
419    }
420    let fraction = len as f64 / genome_size as f64;
421    let lower = name.to_ascii_lowercase();
422
423    if lower == "chrm"
424        || lower == "mt"
425        || lower == "chrmt"
426        || lower == "m"
427        || lower.contains('_')
428        || lower.contains("random")
429        || lower.contains("un")
430        || fraction < 0.001
431    {
432        return 2;
433    }
434
435    if (0.02..=0.25).contains(&fraction) {
436        0
437    } else if fraction > 0.25 {
438        1
439    } else {
440        2
441    }
442}

455        path: &std::path::Path,
456        chroms: HashMap<String, u32>,
457        intervals: Vec<(String, u32, u32, f32)>,
458    ) -> anyhow::Result<()> {
459        let iter = intervals.iter().map(|(c, s, e, v)| {
460            (
461                c.as_str(),
462                Value {
463                    start: *s,
464                    end: *e,
465                    value: *v,
466                },
467            )
468        });
469        let bed_iter = BedParserStreamingIterator::wrap_infallible_iter(iter, true);
470        let writer = BigWigWrite::create_file(path, chroms)?;
471        let runtime = tokio::runtime::Builder::new_multi_thread()
472            .worker_threads(1)
473            .build()?;
474        writer.write(bed_iter, runtime)?;
475        Ok(())
476    }

775    /// Build a chromosome where the minimum appears after N leading intervals
776    /// of a higher value.  Used to probe the early-exit heuristic.
777    ///
778    /// Layout (bin_size=10, all on chr5):
779    ///   [0, 10)          → leading_val   (repeated `leading_n` times)
780    ///   [leading_n*10, …) → true_min     (one bin)
781    fn make_late_minimum_bigwig(
782        dir: &tempfile::TempDir,
783        leading_n: u32,
784        leading_val: f32,
785        true_min: f32,
786        bin_size: u32,
787    ) -> anyhow::Result<std::path::PathBuf> {
788        let path = dir.path().join("late_min.bw");
789        let total_bins = leading_n + 1;
790        let chrom_len = total_bins * bin_size;
791        let mut chroms = HashMap::new();
792        // Two chromosomes so the cross-chrom confidence logic can fire.
793        chroms.insert("chr5".to_string(), chrom_len);
794        chroms.insert("chr7".to_string(), chrom_len);
795
796        let mut intervals: Vec<(String, u32, u32, f32)> = Vec::new();
797        for chrom in &["chr5", "chr7"] {
798            for i in 0..leading_n {
799                intervals.push((
800                    chrom.to_string(),
801                    i * bin_size,
802                    (i + 1) * bin_size,
803                    leading_val,
804                ));
805            }
806            // true minimum is the very last bin
807            intervals.push((
808                chrom.to_string(),
809                leading_n * bin_size,
810                (leading_n + 1) * bin_size,
811                true_min,
812            ));
813        }
814        make_bigwig(&path, chroms, intervals)?;
815        Ok(path)
816    }

818    /// The heuristic can miss a minimum that appears only after both a stable streak
819    /// AND a second distinct value have been seen.  Concrete failure pattern:
820    ///   [3.0]*A  [2.0]*(streak+5)  [1.0]*1
821    /// After A intervals of 3.0 the streak counter is A (min2 still MAX → no break).
822    /// The first 2.0 resets the streak; after streak+5 more 2.0s both conditions are
823    /// satisfied and we break before seeing 1.0.
824    #[test]
825    fn test_heuristic_misses_late_minimum() -> anyhow::Result<()> {
826        let streak: u32 = 8;
827        let config = InferScaleConfig {
828            min_stable_streak: streak,
829            max_starts: 512,
830        };
831        let full = full_scan_config();
832
833        // Layout per chromosome: [3.0]*5, [2.0]*(streak+5), [1.0]*1
834        let dir = tempfile::tempdir()?;
835        let path = dir.path().join("late.bw");
836        let bin: u32 = 10;
837        let a: u32 = 5;
838        let b: u32 = streak + 5;
839        let chrom_len = (a + b + 1) * bin;
840        let mut chroms = HashMap::new();
841        chroms.insert("chr5".to_string(), chrom_len);
842        chroms.insert("chr7".to_string(), chrom_len);
843
844        let mut intervals: Vec<(String, u32, u32, f32)> = Vec::new();
845        for chrom in &["chr5", "chr7"] {
846            for i in 0..a {
847                intervals.push((chrom.to_string(), i * bin, (i + 1) * bin, 3.0));
848            }
849            for i in a..(a + b) {
850                intervals.push((chrom.to_string(), i * bin, (i + 1) * bin, 2.0));
851            }
852            intervals.push((chrom.to_string(), (a + b) * bin, (a + b + 1) * bin, 1.0));
853        }
854        make_bigwig(&path, chroms, intervals)?;
855
856        let r_full = infer_scale_factor(&path, &full)?;
857        let r_heur = infer_scale_factor(&path, &config)?;
858
859        // Full scan finds the true minimum.
860        assert!(
861            (r_full.min_val - 1.0).abs() < 1e-6,
862            "full scan min_val should be 1.0, got {}",
863            r_full.min_val
864        );
865        // Heuristic with small streak misses the late minimum — documents the limitation.
866        assert!(
867            r_heur.min_val > 1.0 + 1e-6,
868            "heuristic with streak={streak} should miss the late minimum (got {}); \
869             increase min_stable_streak or use u32::MAX to disable",
870            r_heur.min_val
871        );
872        Ok(())
873    }

875    /// With a sufficiently large streak the heuristic matches the full scan.
876    /// This is the normal operating regime for real BigWig files where a single-read
877    /// bin is rare but not exclusively clustered at the chromosome end.
878    #[test]
879    fn test_heuristic_matches_full_scan_minimum_not_late() -> anyhow::Result<()> {
880        // Minimum is the FIRST bin; all subsequent bins are higher.
881        // Any streak ≥ 1 will find the correct minimum.
882        let dir = tempfile::tempdir()?;
883        let path = make_late_minimum_bigwig(&dir, 0, 2.0, 1.0, 10)?;
884        // With 0 leading intervals the "late_minimum" fixture places 1.0 first.
885        // Re-use the fixture but build a simpler version: min first, rest larger.
886        let path2 = dir.path().join("min_first.bw");
887        {
888            let mut chroms = HashMap::new();
889            chroms.insert("chr5".to_string(), 200u32);
890            chroms.insert("chr7".to_string(), 200u32);
891            let mut ivs: Vec<(String, u32, u32, f32)> = Vec::new();
892            for chrom in &["chr5", "chr7"] {
893                ivs.push((chrom.to_string(), 0, 10, 1.0_f32));
894                for i in 1..20u32 {
895                    ivs.push((chrom.to_string(), i * 10, (i + 1) * 10, 2.0));
896                }
897            }
898            make_bigwig(&path2, chroms, ivs)?;
899        }
900        let _ = path; // suppress warning
901
902        let r_full = infer_scale_factor(&path2, &full_scan_config())?;
903        let r_heur = infer_scale_factor(
904            &path2,
905            &InferScaleConfig {
906                min_stable_streak: 5,
907                ..Default::default()
908            },
909        )?;
910
911        assert!(
912            (r_full.min_val - r_heur.min_val).abs() < 1e-9,
913            "heuristic and full scan should agree when minimum is not late: full={} heur={}",
914            r_full.min_val,
915            r_heur.min_val
916        );
917        assert_eq!(r_full.norm_method, r_heur.norm_method);
918        Ok(())
919    }
